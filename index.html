using System; using System.IO; using System.Collections.Generic; using UnityEngine; /// /// AvatarHub2D - single-file 2D avatar editor + two simple mini-games. /// Attach to a GameObject. Create child GameObjects (SpriteRenderer) for each layer. /// Configure layers & renderers in Inspector. Assign itemSprite and obstacleSprite for games. /// /// Controls: /// - In Editor/Hub UI (left): edit avatar, save/load, export/import JSON. /// - In games: Move avatar with Left/Right arrows or A/D. /// - Back to Hub from game via UI button. /// /// This is a compact demo — extend as needed. /// [Serializable] public class AvatarLayer { public string layerName = "Layer"; public Sprite[] options = new Sprite[0]; public bool tintable = true; public Color color = Color.white; [NonSerialized] public int selectedIndex = 0; } [Serializable] public class AvatarData { public int[] selectedIndices; public Color[] colors; } public class AvatarHub2D : MonoBehaviour { public AvatarLayer[] layers = new AvatarLayer[0]; public SpriteRenderer[] renderers = new SpriteRenderer[0]; [Header("Game sprites (assign simple sprites)")] public Sprite itemSprite; public Sprite obstacleSprite; [Header("Game settings")] public float playerSpeed = 6f; public float collectibleFallSpeed = 3f; public float obstacleSpeed = 2.5f; public float spawnInterval = 1.0f; public int startingLives = 3; [Header("Save")] public string playerPrefsKey = "avatar_json"; public string saveFileName = "avatar.json"; enum Mode { Hub, Editor, Game1_Collect, Game2_Dodge } Mode mode = Mode.Hub; // runtime state private Vector2 scroll = Vector2.zero; private float spawnTimer = 0f; private int score = 0; private int lives = 0; private List spawned = new List(); // basic play area (world coordinates) private Rect playArea = new Rect(-6f, -4f, 12f, 8f); // x,y,width,height // Use the same GameObject transform as player root (script attached to it). private Transform playerTransform; private float collisionRadius = 0.6f; [Serializable] class SpawnInstance { public GameObject go; public Vector2 velocity; public bool isCollectible; } void Start() { playerTransform = this.transform; if (renderers == null) renderers = new SpriteRenderer[0]; // initialize selected indices for (int i = 0; i < layers.Length; i++) { var L = layers[i]; if (L.options == null || L.options.Length == 0) L.selectedIndex = -1; else L.selectedIndex = Mathf.Clamp(L.selectedIndex, 0, L.options.Length - 1); } ApplyAll(); } void Update() { // simple movement when in game modes if (mode == Mode.Game1_Collect || mode == Mode.Game2_Dodge) { float h = Input.GetAxis("Horizontal"); Vector3 p = playerTransform.position; p.x += h * playerSpeed * Time.deltaTime; // clamp inside play area horizontally float minX = playArea.xMin + 0.5f; float maxX = playArea.xMax - 0.5f; p.x = Mathf.Clamp(p.x, minX, maxX); playerTransform.position = p; // spawn logic spawnTimer += Time.deltaTime; if (spawnTimer >= spawnInterval) { spawnTimer = 0f; if (mode == Mode.Game1_Collect) SpawnCollectible(); else SpawnObstacle(); } // move spawned and check collisions for (int i = spawned.Count - 1; i >= 0; i--) { var s = spawned[i]; if (s == null || s.go == null) { spawned.RemoveAt(i); continue; } s.go.transform.position += (Vector3)(s.velocity * Time.deltaTime); // remove when out of play area Vector2 pos = s.go.transform.position; if (pos.y < playArea.yMin - 2f || pos.y > playArea.yMax + 2f || pos.x < playArea.xMin - 3f || pos.x > playArea.xMax + 3f) { Destroy(s.go); spawned.RemoveAt(i); continue; } // collision with player (distance-based) if (Vector2.Distance(s.go.transform.position, (Vector2)playerTransform.position) < collisionRadius) { if (s.isCollectible) { score += 1; } else { lives -= 1; } Destroy(s.go); spawned.RemoveAt(i); } } // check lose condition (for Game2) if (mode == Mode.Game2_Dodge && lives <= 0) { // back to hub on loss mode = Mode.Hub; CleanupSpawned(); } } } void SpawnCollectible() { if (itemSprite == null) return; float x = UnityEngine.Random.Range(playArea.xMin + 0.5f, playArea.xMax - 0.5f); Vector2 pos = new Vector2(x, playArea.yMax + 1.0f); GameObject go = new GameObject("Collectible"); go.transform.position = pos; var sr = go.AddComponent(); sr.sprite = itemSprite; sr.sortingOrder = 100; var s = new SpawnInstance(); s.go = go; s.velocity = new Vector2(0f, -collectibleFallSpeed); s.isCollectible = true; spawned.Add(s); } void SpawnObstacle() { if (obstacleSprite == null) return; // Spawn from top moving down-left or down-right float x = UnityEngine.Random.Range(playArea.xMin + 0.5f, playArea.xMax - 0.5f); Vector2 pos = new Vector2(x, playArea.yMax + 1.0f); GameObject go = new GameObject("Obstacle"); go.transform.position = pos; var sr = go.AddComponent(); sr.sprite = obstacleSprite; sr.sortingOrder = 100; var s = new SpawnInstance(); s.go = go; // obstacles can have slight horizontal velocity float vx = UnityEngine.Random.Range(-1f, 1f) * 0.6f; s.velocity = new Vector2(vx, -obstacleSpeed); s.isCollectible = false; spawned.Add(s); } void CleanupSpawned() { for (int i = 0; i < spawned.Count; i++) if (spawned[i] != null && spawned[i].go != null) Destroy(spawned[i].go); spawned.Clear(); } void OnGUI() { // Left panel UI GUILayout.BeginArea(new Rect(10, 10, 360, Screen.height - 20), GUI.skin.box); scroll = GUILayout.BeginScrollView(scroll); GUILayout.Label("Avatar Hub & Mini-Games"); GUILayout.Space(6); GUILayout.BeginHorizontal(); if (GUILayout.Button("Hub", GUILayout.Width(110))) { mode = Mode.Hub; CleanupSpawned(); } if (GUILayout.Button("Editor", GUILayout.Width(110))) { mode = Mode.Editor; CleanupSpawned(); } GUILayout.EndHorizontal(); GUILayout.Space(6); if (mode == Mode.Hub) { GUILayout.Label("Games:"); if (GUILayout.Button("Play Game 1 — Collect (falling items)")) { StartGame1(); } if (GUILayout.Button("Play Game 2 — Dodge (avoid obstacles)")) { StartGame2(); } GUILayout.Space(6); GUILayout.Label("Avatar quick:"); GUILayout.BeginHorizontal(); if (GUILayout.Button("Randomize")) { RandomizeAvatar(); ApplyAll(); } if (GUILayout.Button("Reset")) { ResetAvatar(); ApplyAll(); } GUILayout.EndHorizontal(); GUILayout.Space(6); GUILayout.BeginHorizontal(); if (GUILayout.Button("Save (PlayerPrefs)")) { SaveToPlayerPrefs(); } if (GUILayout.Button("Load (PlayerPrefs)")) { LoadFromPlayerPrefs(); ApplyAll(); } GUILayout.EndHorizontal(); GUILayout.BeginHorizontal(); if (GUILayout.Button("Save to file")) { SaveToFile(); } if (GUILayout.Button("Load from file")) { LoadFromFile(); ApplyAll(); } GUILayout.EndHorizontal(); GUILayout.Space(6); if (GUILayout.Button("Export JSON to clipboard")) { GUIUtility.systemCopyBuffer = ExportToJson(); } if (GUILayout.Button("Import JSON from clipboard")) { ImportFromJson(GUIUtility.systemCopyBuffer); ApplyAll(); } GUILayout.Space(8); GUILayout.Label("Play area (world coords): " + playArea.ToString()); } else if (mode == Mode.Editor) { GUILayout.Label("Avatar Editor"); if (layers == null || layers.Length == 0) { GUILayout.Label("No layers defined. Configure in Inspector."); } else { for (int i = 0; i < layers.Length; i++) { DrawLayerEditor(i); GUILayout.Space(6); } } } else // game modes { GUILayout.Label("In Game"); GUILayout.Label("Mode: " + (mode == Mode.Game1_Collect ? "Collect" : "Dodge")); GUILayout.Label("Score: " + score); if (mode == Mode.Game2_Dodge) GUILayout.Label("Lives: " + lives); if (GUILayout.Button("Back to Hub")) { mode = Mode.Hub; CleanupSpawned(); } GUILayout.Space(6); GUILayout.Label("Use A/D or Left/Right to move."); } GUILayout.EndScrollView(); GUILayout.EndArea(); // Game HUD drawn near top center if (mode == Mode.Game1_Collect || mode == Mode.Game2_Dodge) { GUILayout.BeginArea(new Rect(Screen.width / 2 - 120, 6, 240, 90), GUI.skin.box); GUILayout.Label((mode == Mode.Game1_Collect ? "Game 1 — Collect" : "Game 2 — Dodge"), GUI.skin.label); GUILayout.Label("Score: " + score); if (mode == Mode.Game2_Dodge) GUILayout.Label("Lives: " + lives); GUILayout.EndArea(); } } void DrawLayerEditor(int i) { var layer = layers[i]; GUILayout.BeginVertical(GUI.skin.box); GUILayout.Label(string.Format("{0}", layer.layerName)); GUILayout.BeginHorizontal(); if (GUILayout.Button("Prev", GUILayout.Width(70))) { ChangeIndex(i, -1); ApplyAll(); } GUILayout.Label((layer.options != null && layer.options.Length > 0 && layer.selectedIndex >= 0) ? string.Format("{0} / {1}", Mathf.Max(0, layer.selectedIndex + 1), layer.options.Length) : "No options", GUILayout.Width(120)); if (GUILayout.Button("Next", GUILayout.Width(70))) { ChangeIndex(i, +1); ApplyAll(); } GUILayout.EndHorizontal(); if (layer.tintable) { GUILayout.Space(4); GUILayout.Label("Tint (RGB)"); layer.color.r = GUILayout.HorizontalSlider(layer.color.r, 0f, 1f); layer.color.g = GUILayout.HorizontalSlider(layer.color.g, 0f, 1f); layer.color.b = GUILayout.HorizontalSlider(layer.color.b, 0f, 1f); GUILayout.BeginHorizontal(); GUILayout.Label("R:" + Mathf.RoundToInt(layer.color.r * 255), GUILayout.Width(60)); GUILayout.Label("G:" + Mathf.RoundToInt(layer.color.g * 255), GUILayout.Width(60)); GUILayout.Label("B:" + Mathf.RoundToInt(layer.color.b * 255), GUILayout.Width(60)); GUILayout.EndHorizontal(); } else { GUILayout.Label("Tint disabled for this layer."); } GUILayout.EndVertical(); } void ChangeIndex(int layerIndex, int delta) { var layer = layers[layerIndex]; if (layer.options == null || layer.options.Length == 0) { layer.selectedIndex = -1; return; } int len = layer.options.Length; int idx = layer.selectedIndex; if (idx < 0) idx = 0; idx = (idx + delta) % len; if (idx < 0) idx += len; layer.selectedIndex = idx; } void ApplyAll() { int n = Mathf.Min(layers.Length, renderers.Length); for (int i = 0; i < n; i++) { var layer = layers[i]; var rend = renderers[i]; if (rend == null) continue; if (layer.options != null && layer.options.Length > 0 && layer.selectedIndex >= 0) rend.sprite = layer.options[layer.selectedIndex]; else rend.sprite = null; if (layer.tintable) rend.color = layer.color; else rend.color = Color.white; } } void RandomizeAvatar() { System.Random r = new System.Random(); for (int i = 0; i < layers.Length; i++) { var layer = layers[i]; if (layer.options != null && layer.options.Length > 0) layer.selectedIndex = r.Next(0, layer.options.Length); if (layer.tintable) layer.color = new Color((float)r.NextDouble(), (float)r.NextDouble(), (float)r.NextDouble(), 1f); } } void ResetAvatar() { for (int i = 0; i < layers.Length; i++) { var layer = layers[i]; if (layer.options != null && layer.options.Length > 0) layer.selectedIndex = 0; if (layer.tintable) layer.color = Color.white; } } string ExportToJson() { AvatarData data = new AvatarData(); data.selectedIndices = new int[layers.Length]; data.colors = new Color[layers.Length]; for (int i = 0; i < layers.Length; i++) { data.selectedIndices[i] = (layers[i].selectedIndex >= 0) ? layers[i].selectedIndex : -1; data.colors[i] = layers[i].color; } string json = JsonUtility.ToJson(data, true); return json; } void ImportFromJson(string json) { if (string.IsNullOrEmpty(json)) { Debug.LogWarning("Import JSON is empty."); return; } try { AvatarData data = JsonUtility.FromJson(json); if (data != null && data.selectedIndices != null) { int m = Mathf.Min(data.selectedIndices.Length, layers.Length); for (int i = 0; i < m; i++) { layers[i].selectedIndex = data.selectedIndices[i]; } } if (data != null && data.colors != null) { int m = Mathf.Min(data.colors.Length, layers.Length); for (int i = 0; i < m; i++) { layers[i].color = data.colors[i]; } } ApplyAll(); } catch (Exception e) { Debug.LogWarning("Failed to import avatar JSON: " + e.Message); } } void SaveToPlayerPrefs() { string json = ExportToJson(); PlayerPrefs.SetString(playerPrefsKey, json); PlayerPrefs.Save(); Debug.Log("Avatar saved to PlayerPrefs."); } void LoadFromPlayerPrefs() { if (!PlayerPrefs.HasKey(playerPrefsKey)) { Debug.LogWarning("No avatar in PlayerPrefs."); return; } string json = PlayerPrefs.GetString(playerPrefsKey); ImportFromJson(json); ApplyAll(); Debug.Log("Avatar loaded from PlayerPrefs."); } void SaveToFile() { try { string path = Path.Combine(Application.persistentDataPath, saveFileName); File.WriteAllText(path, ExportToJson()); Debug.Log("Saved avatar to " + path); } catch (Exception e) { Debug.LogWarning("SaveToFile failed: " + e.Message); } } void LoadFromFile() { try { string path = Path.Combine(Application.persistentDataPath, saveFileName); if (!File.Exists(path)) { Debug.LogWarning("Save file not found: " + path); return; } string json = File.ReadAllText(path); ImportFromJson(json); ApplyAll(); Debug.Log("Loaded avatar from " + path); } catch (Exception e) { Debug.LogWarning("LoadFromFile failed: " + e.Message); } } void StartGame1() { mode = Mode.Game1_Collect; score = 0; spawnTimer = 0f; CleanupSpawned(); // position player Vector3 p = playerTransform.position; p.x = 0f; p.y = playArea.yMin + 1.0f; playerTransform.position = p; } void StartGame2() { mode = Mode.Game2_Dodge; score = 0; lives = startingLives; spawnTimer = 0f; CleanupSpawned(); Vector3 p = playerTransform.position; p.x = 0f; p.y = playArea.yMin + 1.0f; playerTransform.position = p; } // Optional: on draw gizmos show play area and player void OnDrawGizmosSelected() { Gizmos.color = Color.cyan; Gizmos.DrawWireCube(new Vector3(playArea.center.x, playArea.center.y, 0f), new Vector3(playArea.width, playArea.height, 0f)); if (Application.isPlaying && playerTransform != null) { Gizmos.color = Color.yellow; Gizmos.DrawWireSphere(playerTransform.position, collisionRadius); } } }
